\begin{frame}{Why? \textit{2025 Edition.}}
  \large
  All the same reasons as for CoAP-over-WS

  \bigskip

  \begin{itemize}
    \item Extend CoAP, especially \textit{all of EDHOC/OSCORE}, to applications with limited APIs

      \bigskip

      “Smart” phone apps, browsers\footnote{\textit{Still Chrome-only; small movement at Mozilla, no known activity for higher-level BLE interface.}}

      \bigskip

    \item \sout{Too lazy to implement IPSP}

      \textit{\ldots and if the alternative is custom GATT, I'd rather take CoAP-over-non-IP.}

    \bigskip

    \item \sout{But with IPSP I'll have to have a full IP stack}

      \textit{\ldots and sometimes options are limited by power over efficient BLE modules.}

    \bigskip

  \item \sout{Explain non-CoAP GATT values to CoAP devices through SCHC}

    \textit{\ldots is still around as an idea, but not so charming without OSCORE.}

  \end{itemize}
\end{frame}

\begin{frame}{Capabilities}\Large
  \framesubtitle{Similar to CoAP-over-UDP}
  \begin{itemize}
    \item Requests and responses in either direction
    \item Concurrent requests (using tokens)
    \item Non-traditional responses (no special-casing needed)
    \item Reliable and unreliable transport
    \item Optional message deduplication
    \item Implicit flow control
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Methods}\Large
  \framesubtitle{Diverging from -UDP due to different lower-layer guarantees\footnote{Reliable transmission, total order}}
  \begin{itemize}
    \item \texttt{M}-bit is single-bit message ID equivalent
    \item \texttt{C}-bit requests confirmation
    \item \texttt{A}-bit mirrors last \texttt{C} message's \texttt{M}-bit
    \item No timer used: GATT's reliability mechanism suffice
  \end{itemize}

\bigskip
